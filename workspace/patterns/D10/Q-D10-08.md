```cypher
// @id Q-D10-08
// @param population:string
MATCH (t:Target {key: $targetKey})
CALL (t) {
  // means asserted to worsen T (D5 goal predicate), non-guideline speech acts
  MATCH (t)<-[:HAS_OBJECT]-(a:Assertion {predicate: 'WORSENS_TARGET'})-[:HAS_SUBJECT]->(m)
  WHERE a.recordedAt <= $asOf AND (a.recordedTo IS NULL OR a.recordedTo > $asOf) AND ($validAt IS NULL OR ((a.validFrom IS NULL OR a.validFrom <= $validAt) AND (a.validTo IS NULL OR a.validTo > $validAt))) AND NOT coalesce(a.speechAct, 'STATES') IN ['CAUTIONS', 'RECOMMENDS']
  RETURN m, 'assertion' AS basis, a.uid AS uid, a.predicate AS detail, a.qualifiers AS scope, null AS certainty, COLLECT { MATCH (a)-[tev:HAS_EVIDENCE]->(tloc:SourceLocator)<-[:HAS_LOCATOR]-(tsnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tsrc:Source)
    RETURN {via: a.uid, role: tev.role, origin: tev.origin, selectorKind: tloc.selectorKind, exact: tloc.exact, page: tloc.page,
            startOffset: tloc.startOffset, contentHash: tsnap.contentHash, retrievedAt: tsnap.retrievedAt, uri: tsrc.canonicalUri} } AS trace
  UNION
  // BellLabs harm assessments about a means and the target
  MATCH (t)<-[:ASSESSES]-(s:Assessment {effectDirection: 'HARM'})-[:ASSESSES]->(m)
  WHERE m <> t AND s.status = 'ACCEPTED' AND s.recordedAt <= $asOf AND (s.recordedTo IS NULL OR s.recordedTo > $asOf)
  RETURN m, 'assessment' AS basis, s.uid AS uid, s.assessmentKind AS detail, s.facets AS scope,
         coalesce(s.certainty, s.verdict) AS certainty, COLLECT { MATCH (s)-[tev:HAS_EVIDENCE]->(tloc:SourceLocator)<-[:HAS_LOCATOR]-(tsnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tsrc:Source)
      RETURN {via: s.uid, role: tev.role, exact: tloc.exact, contentHash: tsnap.contentHash, uri: tsrc.canonicalUri} AS e
    UNION
    MATCH (s)-[:DERIVED_FROM]->(tin:Assertion)-[tev:HAS_EVIDENCE]->(tloc:SourceLocator)<-[:HAS_LOCATOR]-(tsnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tsrc:Source)
      RETURN {via: tin.uid, role: tev.role, exact: tloc.exact, contentHash: tsnap.contentHash, uri: tsrc.canonicalUri} AS e } AS trace
  UNION
  // R-83 guideline branch: goal predicates with a guideline speech act (no RECOMMENDS_AGAINST -> Target)
  MATCH (t)<-[:HAS_OBJECT]-(a:Assertion)-[:HAS_SUBJECT]->(m)
  WHERE a.predicate IN ['WORSENS_TARGET', 'NO_EFFECT_ON_TARGET'] AND a.speechAct IN ['CAUTIONS', 'RECOMMENDS']
    AND a.recordedAt <= $asOf AND (a.recordedTo IS NULL OR a.recordedTo > $asOf) AND ($validAt IS NULL OR ((a.validFrom IS NULL OR a.validFrom <= $validAt) AND (a.validTo IS NULL OR a.validTo > $validAt)))
    AND ($population IS NULL OR any(q IN coalesce(a.qualifiers, []) WHERE q IN ['population:' + $population, 'population:UNSPECIFIED']))
  RETURN m, 'guideline' AS basis, a.uid AS uid, a.predicate + '/' + a.speechAct AS detail, a.qualifiers AS scope, null AS certainty,
         COLLECT { MATCH (a)-[tev:HAS_EVIDENCE]->(tloc:SourceLocator)<-[:HAS_LOCATOR]-(tsnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tsrc:Source)
    RETURN {via: a.uid, role: tev.role, origin: tev.origin, selectorKind: tloc.selectorKind, exact: tloc.exact, page: tloc.page,
            startOffset: tloc.startOffset, contentHash: tsnap.contentHash, retrievedAt: tsnap.retrievedAt, uri: tsrc.canonicalUri} } AS trace
  UNION
  // R-83 population guidance: USE_CONSTRAINED_IN on the means asserted to serve T
  MATCH (t)<-[:HAS_OBJECT]-(g:Assertion)-[:HAS_SUBJECT]->(m)
  WHERE g.predicate IN ['IMPROVES_TARGET', 'INTENDED_FOR_TARGET'] AND g.recordedAt <= $asOf AND (g.recordedTo IS NULL OR g.recordedTo > $asOf)
  WITH DISTINCT m
  MATCH (m)<-[:HAS_SUBJECT]-(a:Assertion {predicate: 'USE_CONSTRAINED_IN'})-[:HAS_OBJECT]->(cond:Condition)
  WHERE a.speechAct IN ['CAUTIONS', 'RECOMMENDS'] AND a.recordedAt <= $asOf AND (a.recordedTo IS NULL OR a.recordedTo > $asOf) AND ($validAt IS NULL OR ((a.validFrom IS NULL OR a.validFrom <= $validAt) AND (a.validTo IS NULL OR a.validTo > $validAt)))
    AND ($population IS NULL OR any(q IN coalesce(a.qualifiers, []) WHERE q IN ['population:' + $population, 'population:UNSPECIFIED']))
  RETURN m, 'guidelinePopulation' AS basis, a.uid AS uid, cond.name AS detail, a.qualifiers AS scope, null AS certainty,
         COLLECT { MATCH (a)-[tev:HAS_EVIDENCE]->(tloc:SourceLocator)<-[:HAS_LOCATOR]-(tsnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tsrc:Source)
    RETURN {via: a.uid, role: tev.role, origin: tev.origin, selectorKind: tloc.selectorKind, exact: tloc.exact, page: tloc.page,
            startOffset: tloc.startOffset, contentHash: tsnap.contentHash, retrievedAt: tsnap.retrievedAt, uri: tsrc.canonicalUri} } AS trace
  UNION
  // D9 recommendations against a means for this goal (R-83: subject = means, goal = qualifier target:<key>)
  MATCH (a:Assertion {predicate: 'RECOMMENDS_AGAINST', speechAct: 'RECOMMENDS'})-[:HAS_SUBJECT]->(m)
  WHERE ('target:' + t.key) IN coalesce(a.qualifiers, []) AND a.recordedAt <= $asOf AND (a.recordedTo IS NULL OR a.recordedTo > $asOf) AND ($validAt IS NULL OR ((a.validFrom IS NULL OR a.validFrom <= $validAt) AND (a.validTo IS NULL OR a.validTo > $validAt)))
  RETURN m, 'recommendedAgainst' AS basis, a.uid AS uid, a.predicate AS detail, a.qualifiers AS scope, null AS certainty,
         COLLECT { MATCH (a)-[tev:HAS_EVIDENCE]->(tloc:SourceLocator)<-[:HAS_LOCATOR]-(tsnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tsrc:Source)
    RETURN {via: a.uid, role: tev.role, origin: tev.origin, selectorKind: tloc.selectorKind, exact: tloc.exact, page: tloc.page,
            startOffset: tloc.startOffset, contentHash: tsnap.contentHash, retrievedAt: tsnap.retrievedAt, uri: tsrc.canonicalUri} } AS trace
  UNION
  // published BLOCKED rows whose report is about T
  MATCH (t)<-[:ASSESSES]-(r:Report)-[:HAS_ITEM]->(i:ReportItem {disposition: 'BLOCKED'})-[:ASSESSES]->(m)
  WHERE r.recordedAt <= $asOf AND (r.recordedTo IS NULL OR r.recordedTo > $asOf)
  MATCH (i)-[:BLOCKED_BY]->(b)
  RETURN m, 'publishedBlock' AS basis, b.uid AS uid, [lbl IN labels(b) WHERE NOT lbl IN ['Entity', 'Occurrence', 'InformationArtifact', 'EvidenceAssessment']][0] AS detail, null AS scope, null AS certainty,
         COLLECT { MATCH (b:UseConstraint)<-[:RESOLVES_TO_CONSTRAINT]-(tua:Assertion)-[tev:HAS_EVIDENCE]->(tloc:SourceLocator)
                         <-[:HAS_LOCATOR]-(tsnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tsrc:Source)
                   RETURN {via: tua.uid, role: tev.role, exact: tloc.exact, contentHash: tsnap.contentHash, uri: tsrc.canonicalUri} AS e
                   UNION
                   MATCH (b:Assessment)-[:DERIVED_FROM]->(tua:Assertion)-[tev:HAS_EVIDENCE]->(tloc:SourceLocator)
                         <-[:HAS_LOCATOR]-(tsnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tsrc:Source)
                   RETURN {via: tua.uid, role: tev.role, exact: tloc.exact, contentHash: tsnap.contentHash, uri: tsrc.canonicalUri} AS e } AS trace
}
WITH m, basis, uid, detail, scope, certainty, trace WHERE size(trace) > 0
WITH m, collect({basis: basis, uid: uid, detail: detail, scope: scope, certainty: certainty, trace: trace}) AS reasons
RETURN [lbl IN labels(m) WHERE NOT lbl IN ['Entity', 'Occurrence', 'InformationArtifact', 'EvidenceAssessment']][0] AS meansKind, m.uid AS meansUid, m.name AS means, reasons, size(reasons) AS nReasons
ORDER BY nReasons DESC
LIMIT 100
```

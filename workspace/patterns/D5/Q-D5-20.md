## Statement

```cypher
// @id Q-D5-20
MATCH (x:Entity {uid: $xUid})
MATCH (ai:Assertion)-[:HAS_SUBJECT]->(x), (ai)-[:HAS_OBJECT]->(t:Target)
WHERE ai.predicate IN ['INTENDED_FOR_TARGET', 'IMPROVES_TARGET'] AND ai.recordedTo IS NULL
MATCH (ae:Assertion {predicate: 'CAUSES_ADVERSE_EFFECT'})-[:HAS_SUBJECT]->(x), (ae)-[:HAS_OBJECT]->(effect:Condition)
WHERE ae.recordedTo IS NULL
// R-70: Condition is a TargetMeasure member; R-42: no join through a shared Identifier
MATCH (t)-[op:OPERATIONALIZED_BY]->(effect)
WHERE op.recordedTo IS NULL
CALL (ae) {
  // TRACE(ae) + the causality framing: CAUTIONS assertions evidenced on the same snapshot as the report
  OPTIONAL MATCH (ae)-[ev:HAS_EVIDENCE]->(loc:SourceLocator)<-[:HAS_LOCATOR]-(snap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(src:Source)
  OPTIONAL MATCH (adj:Adjudication)-[:ASSESSES]->(ae)
    WHERE adj.recordedAt <= $asOf AND (adj.recordedTo IS NULL OR adj.recordedTo > $asOf)
  OPTIONAL MATCH (ae)-[:ASSERTED_BY]->(who)
  OPTIONAL MATCH (ae)-[:WAS_GENERATED_BY]->(act:Activity)
  OPTIONAL MATCH (cv:Assertion {speechAct: 'CAUTIONS'})-[:HAS_EVIDENCE]->(cloc:SourceLocator)<-[:HAS_LOCATOR]-(snap)
    WHERE cv.recordedTo IS NULL
  RETURN collect(DISTINCT {assertedBy: who.uid, role: ev.role, locator: loc.uid, exact: loc.exact,
                           contentHash: snap.contentHash, retrievedAt: snap.retrievedAt, uri: src.canonicalUri,
                           verdict: adj.verdict, missionId: act.missionId, runId: act.runId}) AS trace,
         collect(DISTINCT {assertion: cv.uid, exact: cloc.exact}) AS causalityCaveats
}
RETURN t.key AS target, ai.predicate AS intendedVia, effect.name AS adverseEffect, op.assertionUid AS measureAssertion,
       ae.valueString AS frequencyOrCount, ae.qualifiers AS qualifiers, ae.uid AS adverseEffectAssertion,
       trace, causalityCaveats
```

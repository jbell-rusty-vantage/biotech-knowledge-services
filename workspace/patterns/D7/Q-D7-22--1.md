## Statement

```cypher
// materialized (projection Activity, method d7-evidence-backers/v1); stored as a D10 COMPARISON or PROFILE Report
// DERIVED_FROM every input Assertion, never as D7 edges (read timing: OD-23)
MATCH (c {uid: $compound})                                                          // D1 ChemicalSubstance (or Constituent)
MATCH (st:Study)-[:HAS_ARM]->(:StudyArm)-[:HAS_COMPONENT]->(:InterventionComponent)-[:ADMINISTERS]->(:IngredientMaterial)-[:REALIZES_SUBSTANCE|MATERIAL_CONTAINS]->(c)
CALL (st) {
  MATCH (st)-[sr:HAS_STUDY_ROLE]->(o:Organization) RETURN o, sr.roleKind AS kind, sr.assertionUid AS a
  UNION
  MATCH (st)-[:HAS_STUDY_ROLE {roleKind:'FUNDING_AWARD'}]->(:Award)-[:AWARDED_BY]->(o:Organization) RETURN o, 'AWARD_FUNDER' AS kind, null AS a
  UNION
  MATCH (st)<-[:REPORTS_ON]-(:Publication)<-[:HAS_SUBJECT]-(:Assertion {predicate:'AUTHORED_BY'})-[:HAS_OBJECT]->(au:Person)-[r:HOLDS_ROLE]->(o:Organization)   // R-80
  WHERE r.roleKind IN ['EMPLOYEE','EXECUTIVE_OFFICER','FOUNDER'] RETURN o, 'AUTHOR_EMPLOYER' AS kind, r.assertionUid AS a
}
WITH o, count(DISTINCT st) AS studies, collect(DISTINCT kind) AS kinds, collect(a) AS inputs
WITH collect({o: o, n: studies, kinds: kinds, inputs: inputs}) AS rows, sum(studies) AS total
UNWIND rows AS row
RETURN row.o.name AS backer, row.n AS studies, row.kinds AS kinds, toFloat(row.n) / total AS share, row.inputs AS inputs   // + Herfindahl over shares in the Report
```

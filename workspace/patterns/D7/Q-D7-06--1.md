## Statement

```cypher
// all recommendation occurrences by P with their reachable interests and the disclosure finding.
// The answer is driven by the CONFLICT_RELEVANCE Assessment's DERIVED_FROM inputs; the structural match is a candidate list
// for occurrences no Assessment covers yet, never a verdict.
MATCH (occ:Assertion {assertionKind:'CLAIM', speechAct:'RECOMMENDS'})-[:ASSERTED_BY]->(p:Person {uid: $person})
OPTIONAL MATCH (occ)-[:OCCURS_IN]->(w:MediaWork)                                    // R-19: the container
MATCH (occ)-[:HAS_EVIDENCE {role:'SUPPORTS'}]->(:SourceLocator)<-[:HAS_LOCATOR]-(s:SourceSnapshot)<-[:HAS_SNAPSHOT]-(src:Source)
WITH occ, p, w, collect(DISTINCT src) AS occSources, min(coalesce(s.publishedAt, s.observedAt)) AS snapT
WITH occ, p, w, occSources,
     CASE WHEN w IS NULL THEN snapT ELSE w.publishedAt END AS t,
     CASE WHEN w IS NULL THEN 'SNAPSHOT_FALLBACK' ELSE 'OCCURS_IN_MEDIAWORK' END AS tBasis
OPTIONAL MATCH (occ)-[:HAS_SUBJECT|HAS_OBJECT]->(thing)                             // the recommended thing: subject or object
// (1) assessed interests: the Assessment's interest inputs and the episodes they authorize
OPTIONAL MATCH (cra:Assessment {assessmentKind:'CONFLICT_RELEVANCE'})-[:DERIVED_FROM]->(occ) WHERE cra.recordedTo IS NULL
OPTIONAL MATCH (cra)-[:DERIVED_FROM]->(ia:Assertion) WHERE ia.predicate IN ['HOLDS_ROLE','HOLDS_STAKE','RECEIVES_VALUE_FROM']
OPTIONAL MATCH (p)-[r:HOLDS_ROLE|HOLDS_STAKE|RECEIVES_VALUE_FROM {assertionUid: ia.uid}]->(o:Organization)
// (2) unassessed candidates: same organization or brand family only
OPTIONAL MATCH (p)-[rc:HOLDS_ROLE|HOLDS_STAKE|RECEIVES_VALUE_FROM]->(oc:Organization)-[:PARENT_OF*0..2]-(oc2:Organization)
  WHERE cra IS NULL AND rc.recordedTo IS NULL
    AND (thing = oc2 OR EXISTS { (oc2)-[:OWNS_BRAND]->(:Brand)<-[:BRANDED_AS]-(thing) })
// disclosure in a rendition of the same work (R-19), or in the occurrence's own Source when there is no container
OPTIONAL MATCH (da:Assertion)-[:HAS_EVIDENCE]->(:SourceLocator)<-[:HAS_LOCATOR]-(dsnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(dsrc:Source)
  WHERE (da)-[:HAS_OBJECT]->(o)
    AND ((da.predicate = type(r) AND (da)-[:ASSERTED_BY]->(p))
         OR (da.predicate = 'DISCLOSES_INTEREST' AND (da.polarity IS NULL OR da.polarity <> 'NEGATIVE')
             AND ((da)-[:HAS_SUBJECT]->(p) OR (w IS NOT NULL AND (da)-[:HAS_SUBJECT]->(w)))))
    AND ((w IS NULL AND dsrc IN occSources) OR (w IS NOT NULL AND (dsrc)-[:RENDITION_OF]->(w)))
RETURN occ.uid, w.uid AS container, t, tBasis, thing.name, labels(thing)[0] AS thingKind,
       cra.assessmentSubkind, cra.conclusionCode, cra.facets,
       collect(DISTINCT {family: type(r), kind: coalesce(r.roleKind, r.stakeKind, r.valueKind), org: o.name, from: r.validFrom,
                         to: r.validTo, observedAt: ia.validObservedAt, a: r.assertionUid,
                         disclosure: CASE WHEN da IS NULL THEN null
                                          WHEN dsnap.publishedAt IS NULL OR t IS NULL THEN 'TIMING_UNKNOWN'
                                          WHEN dsnap.publishedAt <= t THEN 'DISCLOSED_IN_CONTAINER'
                                          ELSE 'DISCLOSED_IN_LATER_REVISION' END, disclosureAssertion: da.uid}) AS assessedInterests,
       collect(DISTINCT {family: type(rc), org: oc.name, a: rc.assertionUid}) AS unassessedCandidates
```

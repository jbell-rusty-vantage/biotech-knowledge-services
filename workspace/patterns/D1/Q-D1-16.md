## Statement

```cypher
// @id Q-D1-16
// @param vt:datetime
MATCH (p:ProductVariant {uid: $pUid})-[:HAS_PACKAGE]->(:Package)-[:HAS_LABEL_VERSION]->(:LabelVersion)-[:DECLARES_COMPONENT]->(m:IngredientMaterial)
WHERE m.brandName IS NOT NULL
WITH DISTINCT p, m
OPTIONAL MATCH (a:Assertion)-[:HAS_SUBJECT|HAS_OBJECT]->(m)
WHERE a.predicate IN ['MARKETED_UNDER_MARK', 'SUPPLIES_MATERIAL', 'MANUFACTURES']    // R-68: MANUFACTURES_MATERIAL -> MANUFACTURES (D7 set)
  AND a.recordedTo IS NULL
  AND (a.validFrom IS NULL OR a.validFrom <= $vt) AND (a.validTo IS NULL OR a.validTo > $vt)
OPTIONAL MATCH (a)-[:HAS_SUBJECT|HAS_OBJECT]->(org:Organization)                  // mark owner (object), supplier / manufacturer (subject)
CALL (a) {
  OPTIONAL MATCH (a)-[tev:HAS_EVIDENCE]->(tloc:SourceLocator)<-[:HAS_LOCATOR]-(tsnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tsrc:Source)
  OPTIONAL MATCH (a)-[:ASSERTED_BY]->(twho)
  OPTIONAL MATCH (tadj:Adjudication)-[:ASSESSES]->(a) WHERE tadj.recordedTo IS NULL
  RETURN collect(DISTINCT CASE WHEN tloc IS NULL AND twho IS NULL THEN null ELSE
           {asserter: twho.uid, role: tev.role, selectorKind: tloc.selectorKind, exact: tloc.exact, snapshot: tsnap.uid,
            contentHash: tsnap.contentHash, retrievedAt: tsnap.retrievedAt, source: tsrc.canonicalUri,
            adjudicationKind: tadj.adjudicationKind, verdict: tadj.verdict} END) AS trace
}
RETURN m.uid AS materialUid, m.brandName AS brandName, m.name AS material,
       a.predicate AS predicate,
       CASE a.predicate WHEN 'MARKETED_UNDER_MARK' THEN 'MARK_OWNER' WHEN 'SUPPLIES_MATERIAL' THEN 'SUPPLIER'
                        WHEN 'MANUFACTURES' THEN 'MANUFACTURER' END AS role,
       org.uid AS organizationUid, org.name AS organization, a.qualifiers AS qualifiers,   // SUPPLIES_MATERIAL: exclusivity=, buyer=
       a.validFrom AS validFrom, a.validTo AS validTo, a.status AS status, a.uid AS assertion, trace
ORDER BY brandName, role, validFrom
```

## Statement

```cypher
// @id Q-D9-18
MATCH (f:Frontier {uid: $frontierUid})<-[:ON_FRONTIER]-(ev:Event {eventKind: 'DATA_PRESENTATION'})<-[:HAS_OBJECT]-(rec:Assertion {predicate: 'RECORDING_OF'})-[:HAS_SUBJECT]->(talk:MediaWork)
WHERE rec.recordedTo IS NULL
RETURN ev.uid AS event, ev.name, ev.startedAt, talk.uid AS talk, talk.name AS talkName,
       COLLECT { MATCH (ev)-[inv:INVOLVES]->(pp:Person) WHERE inv.role = 'PRESENTER' AND inv.recordedTo IS NULL RETURN DISTINCT pp.name } AS presenters,
       COLLECT { MATCH (talk)<-[ap:APPEARS_IN]-(sp:Person) WHERE ap.appearanceRole IN ['PRESENTER', 'PANELIST'] AND ap.recordedTo IS NULL RETURN DISTINCT sp.name } AS mediaSpeakers,
       COUNT { MATCH (talk)<-[:OCCURS_IN]-(o:Assertion {assertionKind: 'CLAIM'}) WHERE o.recordedTo IS NULL } AS occurrences,
       COLLECT { MATCH (talk)<-[:RENDITION_OF]-(s:Source) RETURN s.canonicalUri } AS recordings,
       COLLECT { MATCH (rec)-[tEv:HAS_EVIDENCE]->(tLoc:SourceLocator)<-[:HAS_LOCATOR]-(tSnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tSrc:Source)
      WHERE tEv.role IN ['SUPPORTS', 'QUALIFIES']
      OPTIONAL MATCH (tSrc)-[:RENDITION_OF]->(tRw:MediaWork)
      RETURN {role: tEv.role, selectorKind: tLoc.selectorKind, exact: tLoc.exact, startOffset: tLoc.startOffset, endOffset: tLoc.endOffset,
              mediaStartSeconds: tLoc.mediaStartSeconds, mediaEndSeconds: tLoc.mediaEndSeconds, speakerLabel: tLoc.speakerLabelInSource,
              contentHash: tSnap.contentHash, retrievedAt: tSnap.retrievedAt, captureCompleteness: tSnap.captureCompleteness,
              uri: tSrc.canonicalUri, genre: tSrc.genre, renditionOfWork: tRw.uid,
              captureFidelity: COLLECT { MATCH (tAdj:Adjudication {adjudicationKind: 'CAPTURE_FIDELITY'})-[:ASSESSES]->(rec) WHERE tAdj.recordedTo IS NULL
                                         RETURN tAdj {.uid, .verdict, .policyOutcome} }} } AS recordingOfTrace
```

## Statement

```cypher
// @id Q-D9-08
MATCH (x:Entity {uid: $compoundUid})<-[:HAS_SUBJECT|HAS_OBJECT]-(:Assertion {assertionKind: 'CLAIM'})-[:OCCURRENCE_OF]->(c:Claim)
WITH DISTINCT c
MATCH (c)<-[:OCCURRENCE_OF]-(o:Assertion {assertionKind: 'CLAIM'})-[:OCCURS_IN]->(w:MediaWork)
WHERE w.publishedAt >= datetime() - duration({months: $n}) AND o.recordedTo IS NULL
MATCH (o)-[:ASSERTED_BY]->(a)
OPTIONAL MATCH (o)-[:DERIVED_FROM]->(prev:Assertion)
WITH c, o, w, a, prev ORDER BY w.publishedAt
RETURN c.uid AS claim, c.proposition,
       c.discourseFirstOccurrenceAt, c.discourseFirstHandCount, c.discourseRetellingCount, c.discourseRecutCount,
       c.discourseMethodVersion, c.discourseDerivedAt,
       collect({at: w.publishedAt, occurrence: o.uid, who: a.name, kind: w.workKind, retells: prev.uid, recut: w.republicationKind,
                trace: COLLECT { MATCH (o)-[tEv:HAS_EVIDENCE]->(tLoc:SourceLocator)<-[:HAS_LOCATOR]-(tSnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tSrc:Source)
      WHERE tEv.role IN ['SUPPORTS', 'QUALIFIES']
      OPTIONAL MATCH (tSrc)-[:RENDITION_OF]->(tRw:MediaWork)
      RETURN {role: tEv.role, selectorKind: tLoc.selectorKind, exact: tLoc.exact, startOffset: tLoc.startOffset, endOffset: tLoc.endOffset,
              mediaStartSeconds: tLoc.mediaStartSeconds, mediaEndSeconds: tLoc.mediaEndSeconds, speakerLabel: tLoc.speakerLabelInSource,
              contentHash: tSnap.contentHash, retrievedAt: tSnap.retrievedAt, captureCompleteness: tSnap.captureCompleteness,
              uri: tSrc.canonicalUri, genre: tSrc.genre, renditionOfWork: tRw.uid,
              captureFidelity: COLLECT { MATCH (tAdj:Adjudication {adjudicationKind: 'CAPTURE_FIDELITY'})-[:ASSESSES]->(o) WHERE tAdj.recordedTo IS NULL
                                         RETURN tAdj {.uid, .verdict, .policyOutcome} }} }})[..50] AS spread
```

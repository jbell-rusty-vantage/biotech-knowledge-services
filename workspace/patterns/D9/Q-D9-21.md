## Statement

```cypher
// @id Q-D9-21
MATCH (p:Person)<-[:ASSERTED_BY]-(o:Assertion {assertionKind: 'CLAIM', speechAct: 'RECOMMENDS', predicate: 'RECOMMENDS_USE'})-[:HAS_SUBJECT]->(prod)
WHERE o.recordedTo IS NULL AND (prod:Product OR prod:ChemicalSubstance)
  AND NOT 'CLINICIAN_PUBLIC_COMMENTATOR' IN coalesce(p.publicActorBasis, [])
  AND ((prod:Product AND prod.productKind = 'PRESCRIPTION_DRUG')
       OR EXISTS { MATCH (prod)<-[st:STATUS_OF]-(ls:LegalStatus {statusKind: 'PRESCRIPTION_ONLY'})
                   WHERE st.recordedTo IS NULL AND ($jurisdiction IS NULL OR ls.jurisdiction = $jurisdiction) })   // R-88
MATCH (o)-[:OCCURS_IN]->(w:MediaWork)<-[:HAS_SUBJECT]-(lk:Assertion {predicate: 'LINKS_TO_OFFER'})-[:HAS_OBJECT]->(target)
WHERE lk.recordedTo IS NULL
OPTIONAL MATCH (target:MerchantListing)-[:HAS_OFFER]->(lo:Offer)
WITH p, o, prod, w, lk, coalesce(lo, target) AS offer
MATCH (offer:Offer)-[:SELLER_OF_RECORD]->(seller:Organization)
OPTIONAL MATCH (p)-[tie:HOLDS_ROLE|HOLDS_STAKE]->(seller)
  WHERE tie.recordedTo IS NULL
OPTIONAL MATCH (ia:Assertion {uid: tie.assertionUid})
RETURN p.uid AS recommender, p.name, prod.uid AS item, prod.name AS itemName, seller.uid AS seller, seller.name AS sellerName,
       type(tie) AS tie, tie.validFrom, tie.validTo,
       CASE WHEN tie IS NULL THEN null ELSE CASE
      WHEN w.publishedAt IS NULL THEN 'UNKNOWN'
      WHEN CASE WHEN tie.validFromPrecision IN ['YEAR', 'MONTH', 'DAY'] AND tie.validFrom IS NOT NULL
                THEN datetime.truncate(toLower(tie.validFromPrecision), tie.validFrom) = datetime.truncate(toLower(tie.validFromPrecision), w.publishedAt) ELSE false END
        OR CASE WHEN tie.validToPrecision IN ['YEAR', 'MONTH', 'DAY'] AND tie.validTo IS NOT NULL
                THEN datetime.truncate(toLower(tie.validToPrecision), tie.validTo) = datetime.truncate(toLower(tie.validToPrecision), w.publishedAt) ELSE false END THEN 'UNKNOWN'
      WHEN (tie.validTo IS NOT NULL AND tie.validTo <= w.publishedAt) OR (tie.validFrom IS NOT NULL AND tie.validFrom > w.publishedAt) THEN 'DISJOINT'
      WHEN tie.validFrom IS NOT NULL AND tie.validFrom <= w.publishedAt AND (tie.validTo > w.publishedAt OR (tie.validTo IS NULL AND ia.validObservedAt >= w.publishedAt)) THEN 'OVERLAPS'
      ELSE 'UNKNOWN' END END AS temporalOverlap,
       w.uid AS container, w.publishedAt, COLLECT { MATCH (o)-[tEv:HAS_EVIDENCE]->(tLoc:SourceLocator)<-[:HAS_LOCATOR]-(tSnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tSrc:Source)
      WHERE tEv.role IN ['SUPPORTS', 'QUALIFIES']
      OPTIONAL MATCH (tSrc)-[:RENDITION_OF]->(tRw:MediaWork)
      RETURN {role: tEv.role, selectorKind: tLoc.selectorKind, exact: tLoc.exact, startOffset: tLoc.startOffset, endOffset: tLoc.endOffset,
              mediaStartSeconds: tLoc.mediaStartSeconds, mediaEndSeconds: tLoc.mediaEndSeconds, speakerLabel: tLoc.speakerLabelInSource,
              contentHash: tSnap.contentHash, retrievedAt: tSnap.retrievedAt, captureCompleteness: tSnap.captureCompleteness,
              uri: tSrc.canonicalUri, genre: tSrc.genre, renditionOfWork: tRw.uid,
              captureFidelity: COLLECT { MATCH (tAdj:Adjudication {adjudicationKind: 'CAPTURE_FIDELITY'})-[:ASSESSES]->(o) WHERE tAdj.recordedTo IS NULL
                                         RETURN tAdj {.uid, .verdict, .policyOutcome} }} } AS recommendationTrace, COLLECT { MATCH (lk)-[tEv:HAS_EVIDENCE]->(tLoc:SourceLocator)<-[:HAS_LOCATOR]-(tSnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tSrc:Source)
      WHERE tEv.role IN ['SUPPORTS', 'QUALIFIES']
      OPTIONAL MATCH (tSrc)-[:RENDITION_OF]->(tRw:MediaWork)
      RETURN {role: tEv.role, selectorKind: tLoc.selectorKind, exact: tLoc.exact, startOffset: tLoc.startOffset, endOffset: tLoc.endOffset,
              mediaStartSeconds: tLoc.mediaStartSeconds, mediaEndSeconds: tLoc.mediaEndSeconds, speakerLabel: tLoc.speakerLabelInSource,
              contentHash: tSnap.contentHash, retrievedAt: tSnap.retrievedAt, captureCompleteness: tSnap.captureCompleteness,
              uri: tSrc.canonicalUri, genre: tSrc.genre, renditionOfWork: tRw.uid,
              captureFidelity: COLLECT { MATCH (tAdj:Adjudication {adjudicationKind: 'CAPTURE_FIDELITY'})-[:ASSESSES]->(lk) WHERE tAdj.recordedTo IS NULL
                                         RETURN tAdj {.uid, .verdict, .policyOutcome} }} } AS linkTrace
```

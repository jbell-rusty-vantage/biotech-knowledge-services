## Statement

```cypher
// @id Q-D9-07
MATCH (recut:MediaWork {uid: $workUid})<-[:HAS_SUBJECT]-(rp:Assertion {predicate: 'REPUBLISHES'})-[:HAS_OBJECT]->(orig:MediaWork)
WHERE rp.recordedTo IS NULL
RETURN recut.uid AS recut, recut.republicationKind, recut.publishedAt, orig.uid AS original, orig.publishedAt AS originalPublishedAt,
       rp.uid AS republishes, rp.status, rp.qualifiers, COLLECT { MATCH (rp)-[tEv:HAS_EVIDENCE]->(tLoc:SourceLocator)<-[:HAS_LOCATOR]-(tSnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tSrc:Source)
      WHERE tEv.role IN ['SUPPORTS', 'QUALIFIES']
      OPTIONAL MATCH (tSrc)-[:RENDITION_OF]->(tRw:MediaWork)
      RETURN {role: tEv.role, selectorKind: tLoc.selectorKind, exact: tLoc.exact, startOffset: tLoc.startOffset, endOffset: tLoc.endOffset,
              mediaStartSeconds: tLoc.mediaStartSeconds, mediaEndSeconds: tLoc.mediaEndSeconds, speakerLabel: tLoc.speakerLabelInSource,
              contentHash: tSnap.contentHash, retrievedAt: tSnap.retrievedAt, captureCompleteness: tSnap.captureCompleteness,
              uri: tSrc.canonicalUri, genre: tSrc.genre, renditionOfWork: tRw.uid,
              captureFidelity: COLLECT { MATCH (tAdj:Adjudication {adjudicationKind: 'CAPTURE_FIDELITY'})-[:ASSESSES]->(rp) WHERE tAdj.recordedTo IS NULL
                                         RETURN tAdj {.uid, .verdict, .policyOutcome} }} } AS republishesTrace,
       COLLECT { MATCH (orig)<-[:OCCURS_IN]-(o:Assertion {assertionKind: 'CLAIM'})-[:HAS_EVIDENCE]->(rloc:SourceLocator)<-[:HAS_LOCATOR]-(rsnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(rsrc:Source)-[:RENDITION_OF]->(recut)
                 WHERE o.recordedTo IS NULL
                 RETURN {occurrence: o.uid, exact: rloc.exact, mediaStartSeconds: rloc.mediaStartSeconds,
                         contentHash: rsnap.contentHash, uri: rsrc.canonicalUri} } AS sharedOccurrences
```

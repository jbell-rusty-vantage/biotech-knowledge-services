## Statement

```cypher
// @id Q-D9-11
MATCH (x:Entity {uid: $itemUid})<-[:HAS_SUBJECT]-(o:Assertion {assertionKind: 'CLAIM', predicate: 'CAUSES_ADVERSE_EFFECT', basisKind: 'PERSONAL_EXPERIENCE'})-[:OCCURS_IN]->(w:MediaWork)
WHERE w.workKind IN ['FORUM_THREAD', 'SOCIAL_POST'] AND o.recordedTo IS NULL
  AND 'sourceKind:COMMUNITY' IN coalesce(o.qualifiers, [])
MATCH (o)-[:ASSERTED_BY]->(acct)
OPTIONAL MATCH (o)-[:HAS_OBJECT]->(effect:Condition)
WITH x, effect, o.polarity AS polarity, count(DISTINCT acct) AS asserters, min(w.publishedAt) AS first, collect(DISTINCT w.uid) AS threads,
     collect(DISTINCT {occurrence: o.uid, trace: COLLECT { MATCH (o)-[tEv:HAS_EVIDENCE]->(tLoc:SourceLocator)<-[:HAS_LOCATOR]-(tSnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tSrc:Source)
      WHERE tEv.role IN ['SUPPORTS', 'QUALIFIES']
      OPTIONAL MATCH (tSrc)-[:RENDITION_OF]->(tRw:MediaWork)
      RETURN {role: tEv.role, selectorKind: tLoc.selectorKind, exact: tLoc.exact, startOffset: tLoc.startOffset, endOffset: tLoc.endOffset,
              mediaStartSeconds: tLoc.mediaStartSeconds, mediaEndSeconds: tLoc.mediaEndSeconds, speakerLabel: tLoc.speakerLabelInSource,
              contentHash: tSnap.contentHash, retrievedAt: tSnap.retrievedAt, captureCompleteness: tSnap.captureCompleteness,
              uri: tSrc.canonicalUri, genre: tSrc.genre, renditionOfWork: tRw.uid,
              captureFidelity: COLLECT { MATCH (tAdj:Adjudication {adjudicationKind: 'CAPTURE_FIDELITY'})-[:ASSESSES]->(o) WHERE tAdj.recordedTo IS NULL
                                         RETURN tAdj {.uid, .verdict, .policyOutcome} }} }})[..20] AS reports
RETURN effect.uid AS effect, effect.name, polarity, asserters, first, threads, 'ANECDOTE_NOT_EVIDENCE' AS label, reports,
       COLLECT { MATCH (ss:Assessment {assessmentKind: 'SAFETY_SIGNAL'})-[:ASSESSES]->(x) WHERE ss.recordedTo IS NULL
                 RETURN {assessment: ss.uid, subkind: ss.assessmentSubkind, conclusion: ss.conclusionCode, facets: ss.facets} } AS safetySignals
```

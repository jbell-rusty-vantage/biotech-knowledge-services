## Statement

```cypher
// @id Q-D9-05
MATCH (x:Entity {uid: $compoundUid})<-[:HAS_SUBJECT|HAS_OBJECT]-(o:Assertion {assertionKind: 'CLAIM'})
WHERE o.valueNumber IS NOT NULL AND o.unitCode IS NOT NULL AND o.recordedTo IS NULL
OPTIONAL MATCH (o)<-[:DERIVED_FROM]-(corr:Assertion {assertionKind: 'CLAIM', predicate: 'CORRECTS'})
  WHERE corr.recordedTo IS NULL
OPTIONAL MATCH (qa:Assessment {assessmentKind: 'QUALITY', assessmentSubkind: 'D9.dose-statement-risk'})-[:DERIVED_FROM]->(o)
  WHERE qa.recordedTo IS NULL
RETURN o.uid AS occurrence, o.valueNumber, o.unitCode, corr.valueNumber AS correctedValue, corr.unitCode AS correctedUnit,
       NOT any(q IN coalesce(o.qualifiers, []) WHERE q STARTS WITH 'dose-basis:') AS basisUnspecified,
       qa.uid AS assessment, qa.conclusionCode, qa.facets,
       COLLECT { MATCH (o)-[tEv:HAS_EVIDENCE]->(tLoc:SourceLocator)<-[:HAS_LOCATOR]-(tSnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tSrc:Source)
      WHERE tEv.role IN ['SUPPORTS', 'QUALIFIES']
      OPTIONAL MATCH (tSrc)-[:RENDITION_OF]->(tRw:MediaWork)
      RETURN {role: tEv.role, selectorKind: tLoc.selectorKind, exact: tLoc.exact, startOffset: tLoc.startOffset, endOffset: tLoc.endOffset,
              mediaStartSeconds: tLoc.mediaStartSeconds, mediaEndSeconds: tLoc.mediaEndSeconds, speakerLabel: tLoc.speakerLabelInSource,
              contentHash: tSnap.contentHash, retrievedAt: tSnap.retrievedAt, captureCompleteness: tSnap.captureCompleteness,
              uri: tSrc.canonicalUri, genre: tSrc.genre, renditionOfWork: tRw.uid,
              captureFidelity: COLLECT { MATCH (tAdj:Adjudication {adjudicationKind: 'CAPTURE_FIDELITY'})-[:ASSESSES]->(o) WHERE tAdj.recordedTo IS NULL
                                         RETURN tAdj {.uid, .verdict, .policyOutcome} }} } AS trace, COLLECT { MATCH (corr)-[tEv:HAS_EVIDENCE]->(tLoc:SourceLocator)<-[:HAS_LOCATOR]-(tSnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tSrc:Source)
      WHERE tEv.role IN ['SUPPORTS', 'QUALIFIES']
      OPTIONAL MATCH (tSrc)-[:RENDITION_OF]->(tRw:MediaWork)
      RETURN {role: tEv.role, selectorKind: tLoc.selectorKind, exact: tLoc.exact, startOffset: tLoc.startOffset, endOffset: tLoc.endOffset,
              mediaStartSeconds: tLoc.mediaStartSeconds, mediaEndSeconds: tLoc.mediaEndSeconds, speakerLabel: tLoc.speakerLabelInSource,
              contentHash: tSnap.contentHash, retrievedAt: tSnap.retrievedAt, captureCompleteness: tSnap.captureCompleteness,
              uri: tSrc.canonicalUri, genre: tSrc.genre, renditionOfWork: tRw.uid,
              captureFidelity: COLLECT { MATCH (tAdj:Adjudication {adjudicationKind: 'CAPTURE_FIDELITY'})-[:ASSESSES]->(corr) WHERE tAdj.recordedTo IS NULL
                                         RETURN tAdj {.uid, .verdict, .policyOutcome} }} } AS correctionTrace
```

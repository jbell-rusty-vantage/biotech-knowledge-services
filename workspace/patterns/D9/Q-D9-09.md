## Statement

```cypher
// @id Q-D9-09
MATCH (ret:Assertion {assertionKind: 'CLAIM'})-[:OCCURS_IN]->(:MediaWork {uid: $workUid})
WHERE ret.recordedTo IS NULL
MATCH (ret)-[:HAS_SUBJECT]->(subj)<-[:HAS_SUBJECT]-(span:Assertion)
WHERE span.predicate = ret.predicate AND span.uid <> ret.uid AND span.recordedTo IS NULL
MATCH (span)-[:HAS_EVIDENCE]->(sl:SourceLocator)
RETURN ret.uid AS retelling, ret.valueNumber, span.uid AS span, span.valueNumber AS spanValue, sl.exact,
       abs(coalesce(ret.valueNumber, 0) - coalesce(span.valueNumber, 0)) AS diff,
       EXISTS { (ret)-[:DERIVED_FROM]->(span) } AS citedSpan,
       COLLECT { MATCH (fa:Assessment {assessmentKind: 'RETELLING_FIDELITY'})-[:DERIVED_FROM]->(ret)
                 WHERE fa.recordedTo IS NULL AND ('retelling=' + ret.uid) IN coalesce(fa.facets, [])
                 RETURN {assessment: fa.uid, facets: fa.facets} } AS fidelity,
       COLLECT { MATCH (ret)-[tEv:HAS_EVIDENCE]->(tLoc:SourceLocator)<-[:HAS_LOCATOR]-(tSnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tSrc:Source)
      WHERE tEv.role IN ['SUPPORTS', 'QUALIFIES']
      OPTIONAL MATCH (tSrc)-[:RENDITION_OF]->(tRw:MediaWork)
      RETURN {role: tEv.role, selectorKind: tLoc.selectorKind, exact: tLoc.exact, startOffset: tLoc.startOffset, endOffset: tLoc.endOffset,
              mediaStartSeconds: tLoc.mediaStartSeconds, mediaEndSeconds: tLoc.mediaEndSeconds, speakerLabel: tLoc.speakerLabelInSource,
              contentHash: tSnap.contentHash, retrievedAt: tSnap.retrievedAt, captureCompleteness: tSnap.captureCompleteness,
              uri: tSrc.canonicalUri, genre: tSrc.genre, renditionOfWork: tRw.uid,
              captureFidelity: COLLECT { MATCH (tAdj:Adjudication {adjudicationKind: 'CAPTURE_FIDELITY'})-[:ASSESSES]->(ret) WHERE tAdj.recordedTo IS NULL
                                         RETURN tAdj {.uid, .verdict, .policyOutcome} }} } AS retellingTrace, COLLECT { MATCH (span)-[tEv:HAS_EVIDENCE]->(tLoc:SourceLocator)<-[:HAS_LOCATOR]-(tSnap:SourceSnapshot)<-[:HAS_SNAPSHOT]-(tSrc:Source)
      WHERE tEv.role IN ['SUPPORTS', 'QUALIFIES']
      OPTIONAL MATCH (tSrc)-[:RENDITION_OF]->(tRw:MediaWork)
      RETURN {role: tEv.role, selectorKind: tLoc.selectorKind, exact: tLoc.exact, startOffset: tLoc.startOffset, endOffset: tLoc.endOffset,
              mediaStartSeconds: tLoc.mediaStartSeconds, mediaEndSeconds: tLoc.mediaEndSeconds, speakerLabel: tLoc.speakerLabelInSource,
              contentHash: tSnap.contentHash, retrievedAt: tSnap.retrievedAt, captureCompleteness: tSnap.captureCompleteness,
              uri: tSrc.canonicalUri, genre: tSrc.genre, renditionOfWork: tRw.uid,
              captureFidelity: COLLECT { MATCH (tAdj:Adjudication {adjudicationKind: 'CAPTURE_FIDELITY'})-[:ASSESSES]->(span) WHERE tAdj.recordedTo IS NULL
                                         RETURN tAdj {.uid, .verdict, .policyOutcome} }} } AS spanTrace
ORDER BY diff
```

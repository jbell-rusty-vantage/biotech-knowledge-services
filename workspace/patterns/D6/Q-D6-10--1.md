## Statement

```cypher
// @id Q-D6-10#1
MATCH (p:Product {uid: $product})-[:HAS_VARIANT]->(v:ProductVariant)-[:HAS_PACKAGE]->(pk:Package)-[r:HAS_LABEL_VERSION]->(lv:LabelVersion)
WHERE r.recordedTo IS NULL AND r.validTo IS NULL
MATCH (a:Assertion {uid: r.assertionUid})-[:ASSERTED_BY]->(src)
CALL (lv) {
  MATCH (lv)-[c:DECLARES_COMPONENT]->() WHERE c.recordedTo IS NULL
  MATCH (la:Assertion {uid: c.assertionUid})
  WITH c, la, [q IN coalesce(la.qualifiers, []) WHERE q STARTS WITH 'statement:' | trim(substring(q, size('statement:')))] AS statements
  WHERE size(statements) > 0                                         // R-30: was c.statementVerbatim IS NOT NULL
  RETURN collect({component: head([q IN coalesce(la.qualifiers, []) WHERE q STARTS WITH 'declaredName:' | trim(substring(q, size('declaredName:')))]),
                  statements: statements, lineAssertion: c.assertionUid}) AS perComponent
}
RETURN v.jurisdiction, lv.uid, src.uid AS declaredBy, a.uid AS labelEpisode, lv.containsAllergens, lv.mayContainAllergens,
       lv.allergenStatementVerbatim, lv.asOfStatementVerbatim, perComponent
```

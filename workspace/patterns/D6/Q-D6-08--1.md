## Statement

```cypher
// @id Q-D6-08#1
MATCH (lv:LabelVersion)-[c:DECLARES_COMPONENT]->(t)
WHERE c.recordedTo IS NULL AND c.lineRole = 'BLEND_MEMBER' AND c.amount IS NULL
  AND ($ingredient IS NULL OR t.uid = $ingredient)
MATCH (lv)-[tot:DECLARES_COMPONENT]->() WHERE tot.recordedTo IS NULL AND tot.lineRole = 'BLEND_TOTAL' AND tot.orderIndex = c.parentOrderIndex
MATCH (p:Product)-[:HAS_VARIANT]->(:ProductVariant)-[:HAS_PACKAGE]->(:Package)-[r:HAS_LABEL_VERSION]->(lv)
WHERE r.recordedTo IS NULL AND r.validTo IS NULL
OPTIONAL MATCH (ma:Assertion {uid: c.assertionUid})                 // R-30: member line Assertion
OPTIONAL MATCH (ba:Assertion {uid: tot.assertionUid})               // R-30: blend-total line Assertion
RETURN DISTINCT p.uid, p.name, lv.uid,
       head([q IN coalesce(ma.qualifiers, []) WHERE q STARTS WITH 'declaredName:' | trim(substring(q, size('declaredName:')))]) AS member,
       head([q IN coalesce(ba.qualifiers, []) WHERE q STARTS WITH 'declaredName:' | trim(substring(q, size('declaredName:')))]) AS blend,
       tot.amount AS blendTotal, tot.unitCode, c.assertionUid AS memberLine, tot.assertionUid AS blendLine
```

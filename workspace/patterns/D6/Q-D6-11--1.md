## Statement

```cypher
// @id Q-D6-11#1
MATCH (o:Organization {uid: $org})-[:OWNS_BRAND]->(b:Brand)<-[bb:BRANDED_AS]-(p:Product {productKind: 'PREPARED_FOOD'})
WHERE bb.recordedTo IS NULL
MATCH (p)-[:HAS_VARIANT]->(v:ProductVariant {jurisdiction: $J})-[:HAS_PACKAGE]->(:Package)-[r:HAS_LABEL_VERSION]->(lv:LabelVersion)
WHERE r.recordedTo IS NULL
MATCH (decl:Assertion {uid: r.assertionUid})-[:ASSERTED_BY]->(declarer)
CALL (lv) {
  MATCH (lv)-[c:DECLARES_COMPONENT]->(t) WHERE c.recordedTo IS NULL
  OPTIONAL MATCH (la:Assertion {uid: c.assertionUid})               // R-30: printed line name
  WITH c, t, la ORDER BY c.orderIndex
  RETURN collect({line: head([q IN coalesce(la.qualifiers, []) WHERE q STARTS WITH 'declaredName:' | trim(substring(q, size('declaredName:')))]),
                  order: c.orderIndex, role: c.lineRole, parent: c.parentOrderIndex, amount: c.amount, unit: c.unitCode,
                  dv: c.dailyValuePercent, target: t.uid, lineAssertion: c.assertionUid}) AS lines,
         collect(DISTINCT t) AS targets
}
CALL (p, targets) {                                                  // R-62: HAS_RECIPE_INGREDIENT on the Product or a line's FoodItem target
  UNWIND [p] + [x IN targets WHERE x:FoodItem] AS s
  MATCH (prep:Assertion {predicate: 'HAS_RECIPE_INGREDIENT'})-[:HAS_SUBJECT]->(s)
  WHERE prep.recordedTo IS NULL
  OPTIONAL MATCH (prep)-[:HAS_OBJECT]->(ing:FoodItem)
  RETURN collect(DISTINCT {subject: s.uid, ingredient: ing.uid, ingredientName: ing.name, amount: prep.valueNumber, unit: prep.unitCode,
                           text: coalesce(prep.valueString, head(COLLECT { MATCH (prep)-[:HAS_EVIDENCE]->(pl:SourceLocator) RETURN pl.exact })),
                           assertion: prep.uid}) AS preparation
}
OPTIONAL MATCH (p)-[hr:HAS_REFERENCE_FOOD]->(ref:FoodItem) WHERE hr.recordedTo IS NULL   // D12 seam: reference composition, never merged
RETURN p.name, lv.uid, declarer.uid, declarer = o AS selfDeclared, lv.asOfStatementVerbatim, decl.validObservedAt AS statedAsOf,
       lines, preparation, collect(DISTINCT ref.uid) AS referenceFoods
```
